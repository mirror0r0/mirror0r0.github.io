{{- /* 项目级图片渲染钩子： - 先尝试当作页面资源 / 全局资源解析（解析成功时替换为 RelPermalink）； - 补 width/height 减少布局跳动（CLS），始终 loading=lazy + decoding=async； - 不在构建期做图片转码：压缩在上传端用 sharp 完成（libvips 低内存）， 服务器只有约 700MB 内存，构建期解码大图会直接打爆（已踩过坑）。 */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $res := false -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $path = strings.TrimPrefix "/" $path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $res = . -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $attrs := merge .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy" "decoding" "async") -}} {{- if and $res (eq $res.ResourceType "image") (gt $res.Width 0) -}} {{- $attrs = merge $attrs (dict "width" (printf "%d" $res.Width) "height" (printf "%d" $res.Height)) -}} {{- end -}} {{- /**/ -}}
